cmake_minimum_required(VERSION 3.16)
project(studio_asr LANGUAGES C CXX)
set(WHISPER_SOURCE_DIR "" CACHE PATH "Sorgenti verificati di whisper.cpp v1.7.6")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "Indicare WHISPER_SOURCE_DIR dal bootstrap verificato")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper-build)
add_library(studio_asr STATIC studio_asr.cpp)
target_compile_features(studio_asr PRIVATE cxx_std_11)
target_link_libraries(studio_asr PUBLIC whisper)
target_include_directories(studio_asr PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
option(STUDIO_ASR_TEST "Compila il controllo sulla fixture italiana" OFF)
if(STUDIO_ASR_TEST)
  add_executable(studio_asr_test ../tests/asr/native.cpp)
  target_link_libraries(studio_asr_test PRIVATE studio_asr)
endif()
