cmake_minimum_required(VERSION 3.22)
project(studio_asr_android LANGUAGES C CXX)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(../asr ${CMAKE_CURRENT_BINARY_DIR}/asr)
add_library(studio_asr_jni SHARED jni.cpp)
target_compile_features(studio_asr_jni PRIVATE cxx_std_17)
target_link_libraries(studio_asr_jni PRIVATE studio_asr log)
target_link_options(studio_asr_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
set_target_properties(studio_asr_jni PROPERTIES OUTPUT_NAME studio_asr)
